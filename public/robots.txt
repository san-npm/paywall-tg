# Gategram — https://gategram.app
# Content Signals (https://contentsignals.org/) express AI usage preferences.
# A single group under `*` applies to ALL crawlers (named groups don't inherit,
# so per-bot groups would silently bypass the Disallow list and waste crawl budget).

User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /

# API routes have no search value. Mini App routes remain crawlable so their
# X-Robots-Tag/meta noindex directives can be seen and existing URLs deindexed.
Disallow: /api/

Sitemap: https://gategram.app/sitemap.xml
